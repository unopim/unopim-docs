<script setup lang="ts">
import { onMounted, ref } from 'vue'

const targetHref = ref<string | null>(null)
const redirecting = ref(false)

onMounted(() => {
  const path = window.location.pathname
  const search = window.location.search || ''
  const hash = window.location.hash || ''
  const LATEST = '2.1'
  const LANDING = '/prologue/'

  let m = path.match(/^\/master(\/.*)?$/)
  if (m) {
    const rest = m[1] && m[1] !== '/' ? m[1] : LANDING
    const target = '/' + LATEST + rest + search + hash
    targetHref.value = target
    redirecting.value = true
    window.location.replace(target)
    return
  }

  m = path.match(/^\/(\d+\.\d+)\.x(\/.*)?$/)
  if (m) {
    const rest = m[2] && m[2] !== '/' ? m[2] : LANDING
    const target = '/' + m[1] + rest + search + hash
    targetHref.value = target
    redirecting.value = true
    window.location.replace(target)
    return
  }
})
</script>

<div class="not-found-wrap">
  <div v-if="redirecting" class="custom-block info">
    <p class="custom-block-title">Redirecting…</p>
    <p>
      Taking you to <a :href="targetHref ?? '/'">{{ targetHref }}</a>.
      If you are not redirected automatically,
      <a :href="targetHref ?? '/'">click here</a>.
    </p>
  </div>

  <template v-else>
    <h1 class="not-found-title">404</h1>
    <p class="not-found-lead">The page you are looking for doesn't exist.</p>
    <p>
      Try the <a href="/">home page</a> or the
      <a href="/2.1/prologue/">latest (2.1) docs</a>.
    </p>
  </template>
</div>
